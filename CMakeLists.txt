cmake_minimum_required(VERSION 3.21.1)

set(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "macOS target architecture")
set(CMAKE_OSX_DEPLOYMENT_TARGET "14.4" CACHE STRING "Minimum supported macOS version")

project(Tekst VERSION 0.1.0 LANGUAGES CXX OBJCXX)

execute_process(
    COMMAND git rev-parse --short=7 HEAD
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    OUTPUT_VARIABLE TEKST_GIT_COMMIT
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
)
if(NOT TEKST_GIT_COMMIT)
    set(TEKST_GIT_COMMIT "unknown")
endif()

execute_process(
    COMMAND git describe --tags --abbrev=0
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    OUTPUT_VARIABLE TEKST_DISPLAY_VERSION
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
)
if(TEKST_DISPLAY_VERSION)
    string(REGEX REPLACE "^v" "" TEKST_DISPLAY_VERSION "${TEKST_DISPLAY_VERSION}")
else()
    set(TEKST_DISPLAY_VERSION ${PROJECT_VERSION})
endif()

# CFBundleVersion: the tag's version plus a fourth number that orders its
# pre-releases before it (0.2.0-alpha.2 -> 0.2.0.2, -beta.2 -> 0.2.0.102,
# -rc.2 -> 0.2.0.502, 0.2.0 -> 0.2.0.1000).
set(TEKST_BUNDLE_VERSION "${PROJECT_VERSION}.1000")
if(TEKST_DISPLAY_VERSION MATCHES "^([0-9]+\\.[0-9]+\\.[0-9]+)(-(alpha|beta|rc)\\.([0-9]+))?$")
    set(release_base "${CMAKE_MATCH_1}")
    set(prerelease_kind "${CMAKE_MATCH_3}")
    set(prerelease_number "${CMAKE_MATCH_4}")
    if(prerelease_kind STREQUAL "alpha")
        set(release_build "${prerelease_number}")
    elseif(prerelease_kind STREQUAL "beta")
        math(EXPR release_build "100 + ${prerelease_number}")
    elseif(prerelease_kind STREQUAL "rc")
        math(EXPR release_build "500 + ${prerelease_number}")
    else()
        set(release_build 1000)
    endif()
    set(TEKST_BUNDLE_VERSION "${release_base}.${release_build}")
endif()

if(APPLE AND CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
    set(CMAKE_INSTALL_PREFIX "${CMAKE_BINARY_DIR}/dist" CACHE PATH
        "Directory for the installed macOS application bundle" FORCE)
endif()

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Qt6 6.12 REQUIRED COMPONENTS Quick QuickControls2 QuickDialogs2)
include(cmake/KSyntaxHighlighting.cmake)

qt_standard_project_setup(REQUIRES 6.12)

include(cmake/AppIcons.cmake)

# The shared Rask modules (git submodule): native window, dialogs, appearance.
set(RASK_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(RASK_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(external/rask)

qt_add_executable(tekst
    main.cpp
    ${TEKST_APP_ICON_FILES}
)

qt_add_qml_module(tekst
    URI Tekst
    VERSION 1.0
    # For qmllint and qmlcachegen: where rask's modules are built.
    IMPORT_PATH "${CMAKE_CURRENT_BINARY_DIR}/external/rask/qml"
    QML_FILES
        GoToLine.qml
        Main.qml
        StatusBar.qml
        TextEditor.qml
    SOURCES
        src/Document.h
        src/Document.cpp
        src/Syntaxes.h
        src/Syntaxes.cpp
        src/TextCodec.h
        src/TextCodec.cpp
        src/editor/EditorAccessible.h
        src/editor/EditorAccessible.cpp
        src/editor/EditorView.h
        src/editor/EditorView.cpp
        src/editor/Indentation.h
        src/editor/Indentation.cpp
        src/editor/Matching.h
        src/editor/Matching.cpp
        src/editor/SyntaxHighlighter.h
        src/editor/SyntaxHighlighter.cpp
        src/editor/TextBuffer.h
        src/editor/TextBuffer.cpp
        src/editor/WrapIndex.h
        src/editor/WrapIndex.cpp
        src/editor/TextSearch.h
        src/editor/TextSearch.cpp
)

# Tekst's highlighting themes (generated by tools/make-syntax-themes.py),
# where SyntaxHighlighter's repository looks for them.
qt_add_resources(tekst syntax_themes
    PREFIX "/tekst/syntax"
    BASE assets
    FILES
        assets/themes/tekst-dark.theme
        assets/themes/tekst-light.theme
)

# The generated QML type registrations include the headers by their bare name.
target_include_directories(tekst PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ${CMAKE_CURRENT_SOURCE_DIR}/src/editor)

# Flag padded class layouts in Debug (the member ordering rule in AGENTS.md),
# on our sources only: Qt's generated code and rask's trip it.
set_property(SOURCE main.cpp src/Document.cpp src/Syntaxes.cpp src/TextCodec.cpp src/editor/TextBuffer.cpp src/editor/EditorView.cpp src/editor/EditorAccessible.cpp
    src/editor/SyntaxHighlighter.cpp src/editor/Matching.cpp src/editor/TextSearch.cpp src/editor/Indentation.cpp src/editor/WrapIndex.cpp
    APPEND PROPERTY COMPILE_OPTIONS $<$<CONFIG:Debug>:-Wpadded>)

# Debug builds get a bundle id of their own, so they don't share macOS
# permissions and Launch Services entries with the installed app.
set(TEKST_BUNDLE_ID com.github.rask-project.tekst)
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    string(APPEND TEKST_BUNDLE_ID ".debug")
endif()

set_target_properties(tekst PROPERTIES
    OUTPUT_NAME "Tekst"
    MACOSX_BUNDLE_BUNDLE_NAME "Tekst"
    MACOSX_BUNDLE_GUI_IDENTIFIER ${TEKST_BUNDLE_ID}
    MACOSX_BUNDLE_BUNDLE_VERSION ${TEKST_BUNDLE_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${TEKST_DISPLAY_VERSION}
    MACOSX_BUNDLE_COPYRIGHT "Copyright (c) 2026 Rask Project"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Info.plist.in"
    MACOSX_BUNDLE TRUE
    WIN32_EXECUTABLE TRUE
)

target_compile_definitions(tekst PRIVATE
    TEKST_VERSION="${TEKST_DISPLAY_VERSION}"
    TEKST_GIT_COMMIT="${TEKST_GIT_COMMIT}"
)

target_link_libraries(tekst
    PRIVATE
        Qt6::Quick
        Qt6::QuickControls2
        Qt6::QuickDialogs2
        KF6::SyntaxHighlighting
        Rask::Controls
        Rask::Platform
        Rask::Window
        Rask::Dialogs
        "-framework CoreFoundation"
)

# Before the bundle is signed below: its signature covers the libraries.
rask_embed_libraries(tekst LIBRARIES KF6::SyntaxHighlighting Rask::Controls Rask::Platform Rask::Window Rask::Dialogs)

install(TARGETS tekst
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

if(APPLE)
    # The linker signs the executable alone, leaving Info.plist unbound;
    # sign the build tree's bundle ad hoc.
    add_custom_command(TARGET tekst POST_BUILD
        COMMAND codesign --force --sign - "$<TARGET_BUNDLE_DIR:tekst>"
        COMMENT "Signing Tekst.app ad hoc"
        VERBATIM
    )

    qt_generate_deploy_qml_app_script(
        TARGET tekst
        OUTPUT_SCRIPT deploy_script
    )
    install(SCRIPT ${deploy_script})
endif()

option(TEKST_BUILD_TOOLS "Build the Tekst tools (editor benchmark)" ${PROJECT_IS_TOP_LEVEL})
if(TEKST_BUILD_TOOLS)
    add_subdirectory(tools/editor-bench)
endif()

option(TEKST_BUILD_TESTS "Build the Tekst tests" ${PROJECT_IS_TOP_LEVEL})
if(TEKST_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
